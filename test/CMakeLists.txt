# On 32-bit x86 hosts (e.g. MinGW i686) GCC defaults to -mfpmath=387. The x87 FPU
# evaluates float expressions at 80-bit extended precision and rounds only on
# store, so a result that is exactly 460.0 under SSE can be 459.99999... here --
# and layout truncates rather than rounds, e.g. (int)(dims.width * scale) in
# ChapterHtmlSlimParser. That turns into off-by-one image dimensions and word
# positions, which makes the golden dumps fail locally while CI (x86-64, SSE2)
# passes. Pin the host tests to SSE2 so they reproduce CI bit-for-bit.
# Firmware targets RISC-V and is unaffected either way.
if(CMAKE_SIZEOF_VOID_P EQUAL 4 AND CMAKE_SYSTEM_PROCESSOR MATCHES "(i.86|x86|AMD64)")
  add_compile_options(-msse2 -mfpmath=sse)
endif()

  -Wall
  -Wextra

)

# On MinGW/MSYS2, statically link the C++/GCC runtimes into each test exe.
# Otherwise the binary resolves libstdc++-6.dll / libgcc_s_seh-1.dll from PATH
# at runtime, and an unrelated copy (e.g. Git-for-Windows' mingw64, or the
# MINGW64/MSVCRT build when the exe was compiled with the UCRT64 toolchain)
# gets loaded. The std::string / iostream ABI mismatch then corrupts memory
# and segfaults deep inside libstdc++ on the file/string-heavy tests
# (hyphenation_eval, epub_benchmark). Static linkage removes the ambiguity.
if(MINGW)
  target_link_options(crosspoint_test_common INTERFACE
    -static-libstdc++
    -static-libgcc
    -static
  )
endif()

# ---------------------------------------------------------------------------
# Generated I18n headers
#
# lib/I18n/I18n.h includes "I18nKeys.h", which is generated by scripts/gen_i18n.py
# and gitignored. The firmware build produces it via the `pre:scripts/gen_i18n.py`
# hook in platformio.ini, so on any machine that has ever built firmware the file
# is simply there -- and a host test that needs the real StrId enum compiles
# locally while failing on a clean checkout, which is what CI always is.
#
# Generated into the BUILD tree, never into lib/I18n/. Writing there would defeat
# the firmware build's own freshness check and leave the un-stripped table in
# place, which costs flash silently (the generator only strips unused keys when
# asked, and platformio.ini asks). Keeping the two outputs apart means the host
# tests cannot influence what the firmware ships.
find_package(Python3 COMPONENTS Interpreter REQUIRED)

set(I18N_GENERATED_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated/I18n")
set(I18N_GENERATED_KEYS "${I18N_GENERATED_DIR}/I18nKeys.h")
# The generator writes the string blob alongside the keys. Declared as an OUTPUT below so a test
# that needs the real strings (rather than just the StrId enum) can list it as a source: CMake
# rejects a source that does not exist at CONFIGURE time unless it knows something generates it,
# and these files are gitignored -- so a tree that has ever built configures fine while a clean
# clone, i.e. CI, fails.
set(I18N_GENERATED_STRINGS "${I18N_GENERATED_DIR}/I18nStrings.cpp")

file(GLOB I18N_TRANSLATION_FILES "${REPO_ROOT}/lib/I18n/translations/*.yaml")

add_custom_command(
  OUTPUT ${I18N_GENERATED_KEYS} ${I18N_GENERATED_STRINGS}
  COMMAND ${CMAKE_COMMAND} -E make_directory ${I18N_GENERATED_DIR}
  COMMAND ${Python3_EXECUTABLE} ${REPO_ROOT}/scripts/gen_i18n.py
          ${REPO_ROOT}/lib/I18n/translations ${I18N_GENERATED_DIR}
  DEPENDS ${REPO_ROOT}/scripts/gen_i18n.py ${I18N_TRANSLATION_FILES}
  COMMENT "Generating I18n headers for host tests"
  VERBATIM
)

# A custom TARGET, not just the custom command's output: a target-level dependency
# is the mechanism that reliably orders across directories. (add_dependencies on an
# INTERFACE library would not -- an interface target has no build rule of its own,
# so the ordering does not reach the consumer's compile step.)
#
# A test that pulls in the real <I18n.h> needs BOTH of the lines below:
#   add_dependencies(<target> i18n_generated)
#   target_include_directories(<target> PRIVATE ${I18N_GENERATED_DIR})
add_custom_target(i18n_generated DEPENDS ${I18N_GENERATED_KEYS} ${I18N_GENERATED_STRINGS})
